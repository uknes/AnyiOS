option(ANYIOS_BUILD_FUZZER "Build LLVM libFuzzer target with sanitizers" OFF)
if(ANYIOS_BUILD_FUZZER)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "AnyiOS fuzz target requires Clang")
    endif()
    add_executable(anyios-fuzz tests/fuzz_macho.cpp src/macho.cpp src/universal.cpp src/linkedit.cpp src/symbols.cpp src/fixup_plan.cpp)
    target_include_directories(anyios-fuzz PRIVATE include)
    target_compile_options(anyios-fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(anyios-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()


option(ANYIOS_WITH_DYNARMIC "Enable experimental ARM64 translation backend (Dynarmic 0BSD)" OFF)
if(ANYIOS_WITH_DYNARMIC)
    include(FetchContent)
    set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
    set(DYNARMIC_FRONTENDS "A64" CACHE STRING "" FORCE)
    set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
    set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
    set(DYNARMIC_USE_LLVM OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(
        dynarmic
        GIT_REPOSITORY https://github.com/azahar-emu/dynarmic.git
        GIT_TAG a46601580d5512d324104f985b5f0209dc980ddc
        GIT_SHALLOW FALSE
        GIT_PROGRESS TRUE
    )
    FetchContent_MakeAvailable(dynarmic)

    add_library(anyios_dynarmic_backend STATIC src/dynarmic_backend.cpp)
    target_include_directories(anyios_dynarmic_backend PUBLIC include)
    target_link_libraries(anyios_dynarmic_backend PUBLIC anyios_guest_memory dynarmic)
    add_executable(anyios-app-entry-probe tools/app_entry_probe.cpp)
    target_link_libraries(anyios-app-entry-probe PRIVATE anyios_dynarmic_backend anyios_owned_object anyios_guest_memory anyios_objc_identity anyios_abi_thunk)
    add_executable(anyios-a64-smoke tests/a64_smoke.cpp)
    target_link_libraries(anyios-a64-smoke PRIVATE anyios_guest_memory anyios_owned_object anyios_dynarmic_backend anyios_abi_thunk anyios_libsystem_shim anyios_guest_tls)
    if(BUILD_TESTING)
        add_test(NAME arm64-on-x64 COMMAND anyios-a64-smoke)
    endif()
endif()

    target_link_libraries(anyios-native-a64-smoke PRIVATE anyios_owned_object anyios_guest_memory anyios_abi_thunk)
    if(BUILD_TESTING)

    endif()
endif()
