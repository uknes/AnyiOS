cmake_minimum_required(VERSION 3.20)
project(AnyiOS VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(anyios_macho STATIC src/macho.cpp src/universal.cpp src/linkedit.cpp src/symbols.cpp)
target_include_directories(anyios_macho PUBLIC include)
target_compile_features(anyios_macho PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(anyios_macho PRIVATE /W4)
else()
    target_compile_options(anyios_macho PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

add_executable(anyios-inspect tools/inspect.cpp)
target_link_libraries(anyios-inspect PRIVATE anyios_macho)

add_library(anyios_owned_object STATIC src/object_code.cpp)
target_include_directories(anyios_owned_object PUBLIC include)
target_link_libraries(anyios_owned_object PUBLIC anyios_macho)
target_compile_features(anyios_owned_object PUBLIC cxx_std_20)

add_library(anyios_guest_memory STATIC src/guest_memory.cpp)
target_include_directories(anyios_guest_memory PUBLIC include)
target_compile_features(anyios_guest_memory PUBLIC cxx_std_20)

include(CTest)
if(BUILD_TESTING)
    add_executable(anyios-memory-tests tests/guest_memory_tests.cpp)
    target_link_libraries(anyios-memory-tests PRIVATE anyios_guest_memory)
    add_test(NAME guest-memory COMMAND anyios-memory-tests)
    add_executable(anyios-executable-tests tests/executable_tests.cpp)
    target_link_libraries(anyios-executable-tests PRIVATE anyios_owned_object anyios_guest_memory)
    add_test(NAME macho-static-executable COMMAND anyios-executable-tests)
    add_executable(anyios-object-tests tests/object_code_tests.cpp)
    target_link_libraries(anyios-object-tests PRIVATE anyios_owned_object)
    add_test(NAME macho-owned-object COMMAND anyios-object-tests)
    add_executable(anyios-tests tests/macho_tests.cpp)
    target_link_libraries(anyios-tests PRIVATE anyios_macho)
    add_test(NAME macho-parser COMMAND anyios-tests)
    add_executable(anyios-symbol-tests tests/symbol_tests.cpp)
    target_link_libraries(anyios-symbol-tests PRIVATE anyios_macho)
    add_test(NAME macho-symbols COMMAND anyios-symbol-tests)
    add_executable(anyios-m1-tests tests/m1_tests.cpp)
    target_link_libraries(anyios-m1-tests PRIVATE anyios_macho)
    add_test(NAME macho-universal-linkedit COMMAND anyios-m1-tests)
    add_test(NAME inspector-help COMMAND anyios-inspect --help)
endif()

option(ANYIOS_BUILD_FUZZER "Build LLVM libFuzzer target with sanitizers" OFF)
if(ANYIOS_BUILD_FUZZER)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "AnyiOS fuzz target requires Clang")
    endif()
    add_executable(anyios-fuzz tests/fuzz_macho.cpp src/macho.cpp src/universal.cpp src/linkedit.cpp src/symbols.cpp)
    target_include_directories(anyios-fuzz PRIVATE include)
    target_compile_options(anyios-fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(anyios-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()


option(ANYIOS_WITH_DYNARMIC "Enable experimental ARM64 translation backend (Dynarmic 0BSD)" OFF)
if(ANYIOS_WITH_DYNARMIC)
    include(FetchContent)
    set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
    set(DYNARMIC_FRONTENDS "A64" CACHE STRING "" FORCE)
    set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
    set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
    set(DYNARMIC_USE_LLVM OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(
        dynarmic
        GIT_REPOSITORY https://github.com/azahar-emu/dynarmic.git
        GIT_TAG a46601580d5512d324104f985b5f0209dc980ddc
        GIT_SHALLOW FALSE
        GIT_PROGRESS TRUE
    )
    FetchContent_MakeAvailable(dynarmic)

    add_executable(anyios-a64-smoke tests/a64_smoke.cpp)
    target_link_libraries(anyios-a64-smoke PRIVATE anyios_guest_memory anyios_owned_object dynarmic)
    if(BUILD_TESTING)
        add_test(NAME arm64-on-x64 COMMAND anyios-a64-smoke)
    endif()
endif()
